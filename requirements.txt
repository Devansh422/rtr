# Python dependencies for the Vercel serverless function (api/index.py).
#
# A byte-identical copy lives at api/requirements.txt as a safety net in case
# Vercel's builder prefers a requirements.txt co-located with the function over
# this project-root one. If you change one, change both.
#
# Pins are carried over from backend/requirements.txt, which is a full freeze of
# the development container. Only what the app actually imports at RUNTIME is
# listed here: the serverless bundle has a hard size limit, so dev/test/lint
# tooling (pytest, black, flake8, mypy, isort, alembic) and unused SDKs (boto3,
# google-*, openai, pandas, numpy, stripe, ...) are deliberately left out.
#
# Alembic in particular is absent on purpose: migrations are applied from CI or
# a developer machine (`cd backend && alembic upgrade head`), never by the
# serverless function itself. A function that races to migrate its own schema on
# cold start is how you get a half-migrated database.

fastapi==0.110.1
starlette==0.37.2
pydantic==2.13.4
email-validator==2.3.0      # backs pydantic's EmailStr
python-multipart==0.0.32    # backs UploadFile / File(...) on POST /api/admin/uploads
python-dotenv==1.2.2

dnspython==2.8.0            # required to resolve mongodb+srv:// Atlas URIs

bcrypt==4.1.3               # password hashing
PyJWT==2.13.0               # access tokens

qrcode==7.4.2               # per-ticket QR codes for event check-in (SVG output, no Pillow needed)

httpx==0.28.1               # Brevo, Meilisearch and Gemini clients
# httpx IS required, despite being imported defensively.
#
# This previously said httpx "ships with the FastAPI/starlette stack" and was
# left unpinned on that basis. It does not: `pip install fastapi` brings
# starlette and pydantic, and httpx only arrives with `fastapi[all]` or a test
# extra. Verified by installing this file alone into an empty environment --
# `import httpx` fails.
#
# The failure mode that hid it is the reason it matters. All three call sites
# wrap the import in try/except and degrade to a no-op, so a deployment with
# GEMINI_API_KEY correctly configured answered every question in retrieval-only
# mode, sent no email, and never reached Meilisearch -- with nothing in the logs
# to say why. A dependency whose absence is silent has to be pinned, not assumed.
